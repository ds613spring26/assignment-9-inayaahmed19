# Grouping by book and detecting chapter boundaries:
austen <- austen_books() |>
  group_by(book) |>
  mutate(
    linenumber = row_number(),
    chapter = cumsum(str_detect(text,
                                regex("^chapter [divxlc]",
                                      ignore_case = TRUE)))
  ) |>
  ungroup()

  1. Tokenization

    • Use unnest_tokens() to tokenize text into individual words.
# Here I filtering just for the 'Emma' book for the subsequent analysis
emma <- austen |>
  filter(book == "Emma")

# Using unnest_token & removing stopwords 
tidy_emma <- emma |>
  unnest_tokens(word, text) |>
  anti_join(stop_words)

# NRC joy sentiment analysis and visualization:
nrc_joy <- get_sentiments("nrc") |>
  filter(sentiment == "joy")

emma_nrc <- tidy_emma |>
  inner_join(nrc_joy, by = "word") |>
  count(word, sort = TRUE) |>
  filter(n > 50)
